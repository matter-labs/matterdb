1//! High-level access to database.
2//!
3//! # Overview
4//!
5//! The core type in this module is the [`Access`] trait, which provides ability to access
6//! [indexes](crate::indexes) from the database. The `Access` trait has several implementations:
7//!
8//! - `Access` is implemented for [`RawAccess`]es, that is, types that provide access to the
9//!   entire database. [`Snapshot`](crate::Snapshot), [`Fork`](crate::Fork) and [`ReadonlyFork`](crate::ReadonlyFork)
10//!   fall into this category.
11//! - [`Prefixed`] restricts an access to a single *namespace*.
12//! - [`Migration`](crate::migration::Migration)s are used for data created during [migrations](crate::migration).
13//!   Similar to `Prefixed`, migrations are separated by namespaces.
14//! - [`Scratchpad`](crate::migration::Scratchpad)s can be used for temporary data. They are distinguished by namespaces as well.
15//!
16//! [`AccessExt`] extends [`Access`] and provides helper methods to instantiate indexes. This
17//! is useful in quick-and-dirty testing. For more complex applications, consider deriving
18//! data schema via [`FromAccess`].
19//!
20//! # Guarantees
21//!
22//! - Namespaced accesses (`Prefixed`, `Migration`s and `Scratchpad`s) do not intersect (i.e.,
23//!   do not have common indexes) for different namespaces. They also do not intersect for
24//!   different access types, even for the same namespace; for example, a `Prefixed` access
25//!   can never access an index from a `Migration` or a `Scratchpad` and vice versa.
26//! - For all listed `Access` implementations, different addresses *within* an `Access` correspond
27//!   to different indexes.
28//! - However, if we consider multiple accesses, indexes can alias. For example, an index
29//!   with address `bar` from a `Prefixed<&Fork>` in namespace `foo` can also be accessed via
30//!   address `foo.bar` from the underlying `Fork`.

46/// High-level access to database data.
47///
48/// This trait is not intended to be implemented by the types outside the crate; indeed,
49/// it instantiates several crate-private types. Correspondingly, `Access` methods
50/// rarely need to be used directly; use [its extension trait](AccessExt) instead.
51///
52/// # Examples
53///
54/// `Access` can be used as a bound on structured database objects and their
55/// readonly methods:
56///
57/// ```
58/// use matterdb::{access::Access, ListIndex, MapIndex};
59///
60/// struct Schema<T: Access> {
61///     list: ListIndex<T::Base, u64>,
62///     map: MapIndex<T::Base, str, u64>,
63/// }
64///
65/// impl<T: Access> Schema<T> {
66///     fn get_some_data(&self) -> Option<u64> {
67///         Some(self.list.get(0)? + self.map.get("foo")?)
68///     }
69/// }
70/// ```
71pub trait Access: Clone {
72    /// Raw access serving as the basis for created indexes.
73    type Base: RawAccess;
74
75    /// Gets index metadata at the specified address, or `None` if there is no index.
76    ///
77    /// # Errors
78    ///
79    /// Returns access errors.
80    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError>;
81
82    /// Gets or creates a generic view with the specified address.
83    ///
84    /// # Errors
85    ///
86    /// Returns access errors.
87    fn get_or_create_view(
88        self,
89        addr: IndexAddress,
90        index_type: IndexType,
91    ) -> Result<ViewWithMetadata<Self::Base>, AccessError>;
92
93    /// Returns an iterator over keys in a group with the specified address.
94    ///
95    /// The iterator buffers keys in memory and may become inconsistent for accesses
96    /// based on [`ReadonlyFork`](crate::ReadonlyFork).
97    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
98    where
99        K: BinaryKey + ?Sized;
100}
101
102impl<T: RawAccess> Access for T {
103    type Base = Self;
104
105    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError> {
106        ViewWithMetadata::get_metadata(self, &addr)
107    }
108
109    fn get_or_create_view(
110        self,
111        addr: IndexAddress,
112        index_type: IndexType,
113    ) -> Result<ViewWithMetadata<Self::Base>, AccessError> {
114        ViewWithMetadata::get_or_create(self, &addr, index_type)
115    }
116
117    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
118    where
119        K: BinaryKey + ?Sized,
120    {
121        GroupKeys::new(self, &base_addr)
122    }
123}

125/// Access that prepends the specified prefix to each created view. The prefix is separated
126/// from user-provided names with a dot char `'.'`.
127///
128/// Since the prefix itself cannot contain a dot, `Prefixed` accesses provide namespace
129/// separation. A set of indexes to which `Prefixed` provides access does not intersect
130/// with a set of indexes accessed by a `Prefixed` instance with another prefix. Additionally,
131/// index in `Prefixed` accesses do not intersect with indexes in special-purpose `Access`
132/// implementations ([`Migration`](crate::migration::Migration)s and [`Scratchpad`](crate::migration::Scratchpad)s).
133///
134/// # Examples
135///
136/// ```
137/// use matterdb::{access::{AccessExt, Prefixed}, Database, TemporaryDB};
138///
139/// let db = TemporaryDB::new();
140/// let fork = db.fork();
141/// let prefixed = Prefixed::new("prefixed", &fork);
142/// prefixed.get_list("list").extend(vec![1_u32, 2, 3]);
143/// let same_list = fork.get_list::<_, u32>("prefixed.list");
144/// assert_eq!(same_list.len(), 3);
145/// ```
146#[derive(Debug, Clone)]
147pub struct Prefixed<T> {
148    access: T,
149    prefix: String,
150}
151
152impl<T: RawAccess> Prefixed<T> {
153    /// Creates a new prefixed access.
154    ///
155    /// # Panics
156    ///
157    /// - Will panic if the prefix is not a [valid prefix name](crate::validation::is_valid_index_name_component()).
158    pub fn new(prefix: impl Into<String>, access: T) -> Self {
159        let prefix = prefix.into();
160        assert_valid_name_component(prefix.as_ref());
161        Self { access, prefix }
162    }
163}
164
165impl<T: RawAccess> Access for Prefixed<T> {
166    type Base = T;
167
168    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError> {
169        let prefixed_addr = addr.prepend_name(self.prefix.as_ref());
170        self.access.get_index_metadata(prefixed_addr)
171    }
172
173    fn get_or_create_view(
174        self,
175        addr: IndexAddress,
176        index_type: IndexType,
177    ) -> Result<ViewWithMetadata<Self::Base>, AccessError> {
178        let prefixed_addr = addr.prepend_name(self.prefix.as_ref());
179        self.access.get_or_create_view(prefixed_addr, index_type)
180    }
181
182    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
183    where
184        K: BinaryKey + ?Sized,
185    {
186        let prefixed_addr = base_addr.prepend_name(self.prefix.as_ref());
187        self.access.group_keys(prefixed_addr)
188    }
189}

191/// Access error together with the location information.
192#[derive(Debug, Error)]
193pub struct AccessError {
194    /// Address of the index where the error has occurred.
195    pub addr: IndexAddress,
196    /// Error kind.
197    #[source]
198    pub kind: AccessErrorKind,
199}
200
201impl fmt::Display for AccessError {
202    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
203        // TODO: implement `Display` for `IndexAddress` for human-readable errors
204        write!(formatter, "Error accessing {:?}: {}", self.addr, self.kind)
205    }
206}

208/// Error that can be emitted during accessing an object from the database.
209#[derive(Debug, Error)]
210#[non_exhaustive]
211pub enum AccessErrorKind {
212    /// Index has wrong type.
213    #[error("Wrong index type: expected {:?}, but got {:?}", expected, actual)]
214    WrongIndexType {
215        /// Expected index type.
216        expected: IndexType,
217        /// Actual index type.
218        actual: IndexType,
219    },
220
221    /// Index name is reserved. It's forbidden for user to create indexes with names
222    /// starting with `__` and not containing a dot `.`.
223    #[error("Index name is reserved")]
224    ReservedName,
225
226    /// Index name is empty.
227    #[error("Index name must not be empty")]
228    EmptyName,
229
230    /// Index contains invalid characters.
231    #[error("Invalid characters used in name ({}). Use {}", name, allowed_chars)]
232    InvalidCharsInName {
233        /// Name that contains invalid chars.
234        name: String,
235        /// Characters allowed in name.
236        allowed_chars: &'static str,
237    },
238
239    /// Invalid tombstone location.
240    #[error("Invalid tombstone location. Tombstones can only be created in migrations")]
241    InvalidTombstone,
242
243    /// Custom error.
244    #[error("{0}")]
245    Custom(#[source] anyhow::Error),
246}

248/// Constructs an object atop the database. The constructed object provides access to data
249/// in the DB, akin to an object-relational mapping.
250///
251/// The access to DB can be readonly or read-write, depending on the `T: Access` type param.
252/// Most object should implement `FromAccess<T>` for all `T: Access`.
253///
254/// Simplest `FromAccess` implementors are indexes; it is also implemented for [`Group`](crate::Group).
255/// `FromAccess` can be implemented for more complex *components*. Thus, `FromAccess` can
256/// be used to compose storage objects from simpler ones.
257///
258/// # Examples
259///
260/// Component with two inner indexes. `FromAccess` is automatically derived using
261/// the `matterdb_derive` crate.
262///
263/// ```
264/// use matterdb_derive::FromAccess;
265/// # use matterdb::{
266/// #     access::{Access, AccessExt, AccessError, FromAccess, RawAccessMut},
267/// #     Database, Entry, Group, MapIndex, IndexAddress, TemporaryDB,
268/// # };
269///
270/// #[derive(FromAccess)]
271/// struct InsertOnlyMap<T: Access> {
272///     map: MapIndex<T::Base, str, String>,
273///     len: Entry<T::Base, u64>,
274/// }
275///
276/// impl<T: Access> InsertOnlyMap<T>
277/// where
278///     T::Base: RawAccessMut,
279/// {
280///     fn insert(&mut self, key: &str, value: String) -> bool {
281///         if self.map.contains(key) { return false; }
282///         self.map.put(&key.to_owned(), value);
283///         self.len.set(self.len.get().unwrap_or_default() + 1);
284///         true
285///     }
286/// }
287///
288/// # fn main() -> Result<(), AccessError> {
289/// let db = TemporaryDB::new();
290/// let fork = db.fork();
291/// # {
292/// let mut map = InsertOnlyMap::from_access(&fork, "test".into())?;
293/// map.insert("foo", "FOO".to_owned());
294/// map.insert("bar", "BAR".to_owned());
295/// assert_eq!(map.len.get(), Some(2));
296/// # }
297///
298/// // Components could be used with `Group` out of the box:
299/// let group_of_maps: Group<_, u16, InsertOnlyMap<_>> =
300///     fork.get_group("test_group");
301/// group_of_maps.get(&1).insert("baz", "BAZ".to_owned());
302/// group_of_maps.get(&2).insert("baz", "BUZZ".to_owned());
303/// # assert_eq!(group_of_maps.get(&1).len.get(), Some(1));
304/// # assert_eq!(
305/// #     fork.get_map::<_, str, String>(("test_group.map", &2_u16)).get("baz").unwrap(),
306/// #     "BUZZ"
307/// # );
308/// # Ok(())
309/// # }
310/// ```
311pub trait FromAccess<T: Access>: Sized {
312    /// Constructs the object at the given address.
313    ///
314    /// # Return value
315    ///
316    /// Returns the constructed object.
317    ///
318    /// # Errors
319    ///
320    /// An error should be returned if the object cannot be constructed.
321    fn from_access(access: T, addr: IndexAddress) -> Result<Self, AccessError>;
322
323    /// Constructs the object from the root of the `access`.
324    ///
325    /// The default implementation uses `Self::from_access()` with an empty address.
326    ///
327    /// # Errors
328    ///
329    /// Returns access errors.
330    fn from_root(access: T) -> Result<Self, AccessError> {
331        Self::from_access(access, IndexAddress::default())
332    }
333}

337    use super::{Access, AccessExt, FromAccess, IndexType, Prefixed};

340    #[test]
341    fn prefixed_works() {
342        let db = TemporaryDB::new();
343        let fork = db.fork();
344        {
345            let prefixed = Prefixed::new("test", &fork);
346            let mut list = prefixed.get_list::<_, i32>("foo");
347            list.extend(vec![1, 2, 3]);
348        }
349        {
350            let list = fork.get_list::<_, i32>("test.foo");
351            assert_eq!(list.len(), 3);
352            assert_eq!(list.iter().collect::<Vec<_>>(), vec![1, 2, 3]);
353        }
354        db.merge_sync(fork.into_patch()).unwrap();
355
356        let snapshot = db.snapshot();
357        let snapshot = snapshot.as_ref();
358        let list = snapshot.get_list::<_, i32>("test.foo");
359        assert_eq!(list.len(), 3);
360        assert_eq!(list.iter().collect::<Vec<_>>(), vec![1, 2, 3]);
361
362        let prefixed = Prefixed::new("test", snapshot);
363        let list = prefixed.get_list::<_, i32>("foo");
364        assert_eq!(list.len(), 3);
365        assert_eq!(list.iter().collect::<Vec<_>>(), vec![1, 2, 3]);
366    }

368    #[test]
369    fn prefixed_views_do_not_collide() {
370        let db = TemporaryDB::new();
371        let fork = db.fork();
372        let foo_space = Prefixed::new("foo", &fork);
373        let bar_space = Prefixed::new("bar", &fork);
374        {
375            let mut list = foo_space.get_list("test");
376            list.push("Test".to_owned());
377            let mut other_list = bar_space.get_list("test");
378            other_list.extend(vec![1_u64, 2, 3]);
379
380            assert_eq!(list.len(), 1);
381            assert_eq!(other_list.len(), 3);
382        }
383        db.merge_sync(fork.into_patch()).unwrap();
384
385        let snapshot = db.snapshot();
386        let foo_space = Prefixed::new("foo", snapshot.as_ref());
387        let list = foo_space.get_list::<_, String>("test");
388        assert_eq!(list.get(0), Some("Test".to_owned()));
389        let bar_space = Prefixed::new("bar", snapshot.as_ref());
390        let list = bar_space.get_list::<_, u64>("test");
391        assert_eq!(list.get(0), Some(1_u64));
392
393        // It is possible to create indexes of the different types at the same (relative) address
394        // in the different `Prefixed` instances.
395        let fork = db.fork();
396        let foo_space = Prefixed::new("foo", &fork);
397        foo_space.get_list::<_, u32>(("fam", &1_u32));
398        let bar_space = Prefixed::new("bar", &fork);
399        bar_space.get_map::<_, u32, u32>(("fam", &1_u32));
400        db.merge_sync(fork.into_patch()).unwrap();
401
402        let snapshot = db.snapshot();
403        let view = snapshot
404            .as_ref()
405            .get_or_create_view(("foo.fam", &1_u32).into(), IndexType::List)
406            .unwrap();
407        assert!(!view.is_phantom());
408        let view = snapshot
409            .as_ref()
410            .get_or_create_view(("bar.fam", &1_u32).into(), IndexType::Map)
411            .unwrap();
412        assert!(!view.is_phantom());
413    }
